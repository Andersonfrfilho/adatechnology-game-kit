        if (process.env.LISTONLY) { console.log('[TILE]', ix, iz); continue }
        if (only && !only.has(`${ix},${iz}`)) continue
        const e = jw.bodies.get(t.solid); if (e) { jw.bi.RemoveBody(e.id); n++ }
      }
      console.log('[REMBOX]', n)
    }
    const before = LABELS.map((l) => ACC[l] ?? 0)
    const pf = process.env.PF ? process.env.PF.split(',').map(Number) : []
    const prof1 = pf.includes(f)
    if (prof1) { post('Profiler.enable'); post('Profiler.setSamplingInterval', { interval: 50 }); post('Profiler.start') }
    const s = performance.now(); room.update(dt); const ms = performance.now() - s
    if (prof1) { let pr: Any = null; session.post('Profiler.stop' as Any, (_e: Any, r: Any) => { pr = r.profile }); console.log('[PF]', f, ms.toFixed(1), '\n' + incl(pr, 45).filter((l) => l.startsWith('INCL')).join('\n')) }
    if (process.env.DUMP && f === Number(process.env.DUMP)) {
      const out: string[] = []
      for (const [key, e] of jw.bodies as Any) {
        if (!e.body.IsActive() || e.body.IsStatic()) continue
        const bb = e.body.GetWorldSpaceBounds(); const mn = bb.mMin, mx = bb.mMax
        const sh = e.body.GetShape()
        const p = e.body.GetPosition()
        out.push(`key${key} ext ${(mx.GetX() - mn.GetX()).toFixed(2)}x${(mx.GetY() - mn.GetY()).toFixed(2)}x${(mx.GetZ() - mn.GetZ()).toFixed(2)} pos ${p.GetX().toFixed(1)},${p.GetY().toFixed(2)},${p.GetZ().toFixed(1)} mass ${(1 / e.body.GetMotionProperties().GetInverseMass()).toFixed(1)} partners ${jw.partners.get(key)?.size ?? 0} v ${Math.hypot(e.body.GetLinearVelocity().GetX(), e.body.GetLinearVelocity().GetY(), e.body.GetLinearVelocity().GetZ()).toFixed(2)} st${sh.GetSubType()} ${(() => { try { let x = sh; if (x.GetSubType() === 11) x = jw.J.castObject(x, jw.J.OffsetCenterOfMassShape).GetInnerShape(); const st = x.GetSubType(); if (st === 7 || st === 8) return 'children ' + jw.J.castObject(x, jw.J.CompoundShape).GetNumSubShapes(); return 'inner' + st } catch (er) { return 'err ' + er } })()} user ${e.user}`)
      }
      console.log('[DUMP]', f, '\n' + out.join('\n'))
    }
    const active = jw.system.GetNumActiveBodies(jw.J.EBodyType_RigidBody)
    let info = ''
    if (ms > 60) {
      const kinds = new Map<string, number>()
      for (const e of jw.bodies.values() as Any) { if (!e.body.IsActive()) continue; const sh = e.body.GetShape(); const k = `${e.layer}/${e.body.IsStatic() ? 's' : 'd'}/st${sh.GetSubType()}/${sh.GetType()}`; kinds.set(k, (kinds.get(k) ?? 0) + 1) }
      info = JSON.stringify([...kinds.entries()])
    }
    info += ` val ${CNT.val} per ${CNT.per} add ${CNT.add}`; CNT.val = 0; CNT.per = 0; CNT.add = 0
    rows.push({ f, ms, stage: LABELS.map((l, i) => (ACC[l] ?? 0) - before[i]!), active, info })
  }
  const digest = room.renderer.pieces().map((pc: Any) => `${pc.objectId}:${pc.root.position.x.toFixed(4)},${pc.root.position.y.toFixed(4)},${pc.root.position.z.toFixed(4)}`).sort().join(' ')
  let prof: Any = null
  if (profile) session.post('Profiler.stop' as Any, (_e: Any, r: Any) => { prof = r.profile })
  session.disconnect()
  return { rows, prof, digest, stats }
}

const incl = (prof: Any, top: number): string[] => {
  const byId = new Map<number, Any>(prof.nodes.map((n: Any) => [n.id, n]))
  const dt = prof.timeDeltas as number[]
  const selfById = new Map<number, number>()
  prof.samples.forEach((id: number, i: number) => selfById.set(id, (selfById.get(id) ?? 0) + (dt[i] ?? 0) / 1000))
  const parent = new Map<number, number>()
  for (const n of prof.nodes as Any[]) for (const c of (n.children ?? []) as number[]) parent.set(c, n.id)
  const inc = new Map<string, number>()
  const self = new Map<string, number>()
  for (const [id, t] of selfById) {
    const n0 = byId.get(id)!
    const k0 = `${n0.callFrame.functionName || '(anon)'} ${String(n0.callFrame.url).split('/').slice(-2).join('/')}:${n0.callFrame.lineNumber}`
    self.set(k0, (self.get(k0) ?? 0) + t)
    const seen = new Set<string>()
    for (let cur: number | undefined = id; cur !== undefined; cur = parent.get(cur)) {
      const n = byId.get(cur)!; const k = `${n.callFrame.functionName || '(anon)'} ${String(n.callFrame.url).split('/').slice(-2).join('/')}:${n.callFrame.lineNumber}`
      if (seen.has(k)) continue
      seen.add(k); inc.set(k, (inc.get(k) ?? 0) + t)
    }
  }
  const a = [...inc.entries()].filter(([k]) => !/^(\(root\)|\(program\)|post |once|fn |js-to-wasm|\(anon\) .*probe)/.test(k)).sort((x, y) => y[1] - x[1]).slice(0, top).map(([k, v]) => `INCL ${v.toFixed(1)}ms ${k}`)
  const s = [...self.entries()].sort((x, y) => y[1] - x[1]).slice(0, 15).map(([k, v]) => `SELF ${v.toFixed(1)}ms ${k}`)
  return [...a, ...s]
}
