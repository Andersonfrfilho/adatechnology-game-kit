import { describe, it } from 'vitest'

import { KIT_WORLD } from '@adatechnology/game-physics'
import type { Material, ShapeDesc, Vec3 } from '@adatechnology/game-physics'
import { buildGroundTile } from '../../game-destruction-three/src/engine-ground'
import { createJoltEngine } from '../src/index'

const GROUND: Material = { frictionStatic: 0.7, frictionDynamic: 0.6, restitution: 0, rolling: 0.015 }
const engine = await createJoltEngine()
const relief = (x: number, z: number): number => -0.05 + 0.01 * Math.sin(1.7 * x + 0.3) * Math.cos(1.3 * z) + 0.004 * Math.sin(5 * x + z)
const TILE = 2

const lay = (world: ReturnType<typeof engine.createWorld>, size: number, x0: number, x1: number, z0: number, z1: number): void => {
  for (let ix = x0; ix < x1; ix += size / TILE * TILE) for (let iz = z0; iz < z1; iz += size) {
    const t = buildGroundTile(relief, ix, iz, size)
    const hf: ShapeDesc = { kind: 'heightfield', origin: [ix, 0, iz], cell: size / t.cells, size: t.cells + 1, heights: t.heights }
    world.createBody({ shape: hf, motion: 'static', position: [0, 0, 0], material: GROUND, user: 0 })
    const children: Array<{ shape: ShapeDesc; position: Vec3 }> = []
    for (let i = 0; i < t.solid.boxes.length; i += 6) children.push({ shape: { kind: 'box', half: [t.solid.boxes[i + 3]!, t.solid.boxes[i + 4]!, t.solid.boxes[i + 5]!] }, position: [t.solid.boxes[i]!, t.solid.boxes[i + 1]!, t.solid.boxes[i + 2]!] })
    if (children.length) world.createBody({ shape: { kind: 'compound', children }, motion: 'static', position: [0, 0, 0], material: GROUND, user: 0 })
  }
}

const roll = (size: number, kind: 'ball' | 'box'): { seam: number; inner: number; endV: number } => {
  const world = engine.createWorld({ ...KIT_WORLD })
  lay(world, size, -4, 24, -2, 4)
  const shape: ShapeDesc = kind === 'ball' ? { kind: 'sphere', radius: 0.3 } : { kind: 'box', half: [0.11, 3.75 * 0, 0.11].map((v, i) => (i === 1 ? 0.11 : v)) as unknown as Vec3 }
  const id = world.createBody({ shape, motion: 'dynamic', mass: 50, position: [0, 0.5, 1.001], material: { frictionStatic: 0.7, frictionDynamic: 0.6, restitution: 0 }, user: 1, linearVelocity: [2.5, 0, 0], angularVelocity: [0, 0, kind === 'ball' ? -8.33 : 0] })
  const pose = new Float64Array(7), vel = new Float64Array(6)
  let seam = 0, inner = 0
  for (let s = 0; s < 120 * 6; s++) {
    world.stepFixed(1 / 120)
    world.readPose(id, pose, 0)
    if (s < 40) continue
    const x = pose[0]!
    const r = pose[1]! - (relief(x, pose[2]!) + (kind === 'ball' ? 0.3 : 0.11))
    const near = Math.abs(x / TILE - Math.round(x / TILE)) * TILE < 0.12
    const a = Math.abs(r)
    if (near) seam = Math.max(seam, a); else inner = Math.max(inner, a)
  }
  world.readVelocity(id, vel, 0)
  const out = { seam, inner, endV: vel[0]! }
  world.dispose?.()
  return out
}

describe.skipIf(process.env.PROBE_SEAM !== '1')('rolling across height-field tile seams', () => {
  it('2 m tiles vs 8 m tiles', () => {
    for (const kind of ['ball', 'box'] as const)
      for (const size of [2, 4, 8]) {
        const r = roll(size, kind)
        console.log('[SEAM3]', kind, 'tile', size, 'max |y - support| at seams', (r.seam * 1000).toFixed(2), 'mm, elsewhere', (r.inner * 1000).toFixed(2), 'mm, end v', r.endV.toFixed(3))
      }
  })
})
